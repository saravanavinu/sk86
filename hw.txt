import java.util.*;

public class ProductInventory {
    public static void main(String[] args) {

        Map<String, Integer> products = new HashMap<>();

        // Add Product
        products.put("Laptop", 10);
        products.put("Mobile", 20);
        products.put("Keyboard", 15);

        // Search Product
        System.out.println("Laptop Quantity: " + products.get("Laptop"));

        // Update Quantity
        products.put("Laptop", 25);

        // Delete Product
        products.remove("Keyboard");

        // Display All Products
        System.out.println("Products:");
        for (Map.Entry<String, Integer> entry : products.entrySet()) {
            System.out.println(entry.getKey() + " : " + entry.getValue());
        }
    }
}

import java.util.*;

public class LibraryBook {
    public static void main(String[] args) {

        Map<Integer, String> books = new HashMap<>();

        // Add Book
        books.put(101, "Java Programming");
        books.put(102, "Python Programming");
        books.put(103, "Data Structures");

        // Search Book
        System.out.println("Book: " + books.get(101));

        // Update Book Details
        books.put(101, "Advanced Java Programming");

        // Remove Book
        books.remove(103);

        // Display All Books
        System.out.println("Books:");
        for (Map.Entry<Integer, String> entry : books.entrySet()) {
            System.out.println(entry.getKey() + " : " + entry.getValue());
        }
    }
}
